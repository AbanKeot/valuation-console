Cost of Debt (After-Tax) const costOfDebtAfterTax = inputs.preTaxCostOfDebt * (1 - inputs.taxRate); // 3. Capital Structure Weights const totalCapital = inputs.marketEquity + inputs.totalDebt; const weightOfEquity = inputs.marketEquity / totalCapital; const weightOfDebt = inputs.totalDebt / totalCapital; // 4. WACC Calculation const wacc = (weightOfEquity * costOfEquity) + (weightOfDebt * costOfDebtAfterTax); return { costOfEquity: { value: costOfEquity, formula: `(${inputs.riskFreeRate * 100}% + (${inputs.beta} × ${inputs.marketRiskPremium * 100}%))`, explanation: "Risk-Free Rate + (Beta × Market Risk Premium)" }, costOfDebtAfterTax: { value: costOfDebtAfterTax, formula: `${inputs.preTaxCostOfDebt * 100}% × (1 - ${inputs.taxRate * 100}%)`, explanation: "Pre-tax Cost of Debt × (1 - Tax Rate)" }, weights: { equity: weightOfEquity, debt: weightOfDebt, formula: `Equity: ${inputs.marketEquity} / ${totalCapital} | Debt: ${inputs.totalDebt} / ${totalCapital}` }, wacc: { value: wacc, formula: `(${weightOfEquity.toFixed(4)} × ${costOfEquity.toFixed(4)}) + (${weightOfDebt.toFixed(4)} × ${costOfDebtAfterTax.toFixed(4)})`, explanation: "(Weight of Equity × Cost of Equity) + (Weight of Debt × After-Tax Cost of Debt)" } };
